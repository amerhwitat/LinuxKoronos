cmake_minimum_required(VERSION 3.20)
project(LinuxKoronos VERSION 0.1.0 LANGUAGES CXX)
include(CTest)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(koronos-host src/main.cpp)
target_compile_options(koronos-host PRIVATE -Wall -Wextra -Wpedantic -Werror)
install(TARGETS koronos-host RUNTIME DESTINATION bin)
if(BUILD_TESTING)
  add_test(NAME host-info COMMAND koronos-host info)
  add_test(NAME host-profile COMMAND koronos-host profile)
endif()
